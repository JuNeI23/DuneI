```python
import unittest

from dunei_motion.spring import critical_spring, exit_spring, simulate, spring, zeta

# Durées de stabilisation à ±1 % affichées par le banc pour les réglages validés le 2026-10-13.
BENCH = {
    "state": ((0.50, 1100, 50), 164),
    "reveal": ((1.00, 600, 50), 286),
    "overlay": ((1.00, 400, 44), 402),
    "window": ((0.70, 300, 27), 274),
    "reflow": ((0.80, 500, 40), 269),
    "spectacle": ((1.00, 150, 25), 569),
    "navigate": ((0.30, 600, 30), 186),
}


class SpringTest(unittest.TestCase):
    """Simulation, dérivation de la sortie et ressort réduit."""

    def test_matches_bench(self):
        for name, (params, ms) in BENCH.items():
            with self.subTest(transition=name):
                self.assertEqual(simulate(spring(*params))[0], ms)

    def test_exit_keeps_zeta_and_scales_duration(self):
        s = spring(1, 600, 50)
        e = exit_spring(s, 0.7)
        self.assertEqual(e, {"mass": 1, "stiffness": 1224.5, "dampening": 71.43})
        self.assertAlmostEqual(zeta(e), zeta(s), places=3)
        self.assertAlmostEqual(simulate(e)[0], 0.7 * simulate(s)[0], delta=2)

    def test_critical_spring(self):
        r = critical_spring(100)
        self.assertEqual(r, {"mass": 1, "stiffness": 4406.3, "dampening": 132.76})
        self.assertAlmostEqual(zeta(r), 1, places=3)
        self.assertAlmostEqual(simulate(r)[0], 100, delta=5)

    def test_overshoot_only_below_critical(self):
        self.assertGreater(simulate(spring(0.70, 300, 27))[1], 0)
        self.assertEqual(simulate(spring(1, 600, 50))[1], 0)


if __name__ == "__main__":
    unittest.main()
```

`dunei_motion/spring.py` :

```python
"""Ressort physique — même modèle que Hyprland (hyprutils) et que le banc proto/motion.

Progression normalisée de 0 à 1, masse m, raideur k, amortissement c, temps en
secondes. Les durées sont mesurées par simulation, jamais par une formule approchée.
"""

from __future__ import annotations

import math

STEP = 0.001          # Pas d'intégration en secondes, identique à proto/motion/physics.js.
EPSILON = 0.001       # Seuil d'arrêt de hyprutils, appliqué à l'écart et à la vitesse.
SETTLE = 0.01         # Tolérance de la durée de stabilisation perçue (±1 %).
CRITICAL_WT = 6.638   # Solution de (1 + ωt)·e^(−ωt) = 0,01, reprise du banc.


def spring(mass: float, stiffness: float, dampening: float) -> dict:
    """Ressort arrondi — forme commune à toutes les cibles.

    Entrées : mass, stiffness, dampening.
    Sortie : {mass, stiffness, dampening}, arrondis au centième, au dixième et au centième.
    Erreurs : aucune. Dépendances : aucune.
    """
    return {"mass": round(mass, 2), "stiffness": round(stiffness, 1), "dampening": round(dampening, 2)}


def zeta(s: dict) -> float:
    """Rapport d'amortissement ζ = c / (2·√(k·m)) ; un ressort dépasse sa cible si ζ < 1.

    Entrée : s — ressort aux valeurs strictement positives.
    Sortie : ζ, non arrondi. Erreurs : ValueError si k·m est négatif. Dépendances : math.
    """
    return s["dampening"] / (2 * math.sqrt(s["stiffness"] * s["mass"]))


def simulate(s: dict) -> tuple[int, float]:
    """Réponse indicielle de 0 à 1, intégrée par Euler semi-implicite comme dans le banc.

    Entrée : s — ressort aux valeurs strictement positives.
    Sortie : (durée en ms après laquelle l'écart reste sous 1 %, dépassement maximal en %).
    Erreurs : ZeroDivisionError si la masse est nulle ; la simulation s'arrête à 10 s.
    Dépendances : STEP, EPSILON, SETTLE.
    """
    m, k, c = s["mass"], s["stiffness"], s["dampening"]
    x = v = peak = 0.0
    settle = 0
    for i in range(1, round(10 / STEP) + 1):
        v += (-k * (x - 1) - c * v) / m * STEP
        x += v * STEP
        peak = max(peak, x)
        if abs(x - 1) >= SETTLE:
            settle = i
        if abs(x - 1) < EPSILON and abs(v) < EPSILON:
            break
    return round(settle * STEP * 1000), round(max(0.0, peak - 1) * 100, 2)


def exit_spring(s: dict, ratio: float) -> dict:
    """Ressort de sortie — le même mouvement, accéléré d'un facteur 1/ratio.

    k ÷ ratio² et c ÷ ratio, masse inchangée : ζ est conservé, les durées sont multipliées par ratio.
    Entrées : s — ressort d'entrée ; ratio — exit_ratio, strictement positif.
    Sortie : ressort arrondi. Erreurs : ZeroDivisionError si ratio est nul. Dépendances : spring.
    """
    return spring(s["mass"], s["stiffness"] / ratio**2, s["dampening"] / ratio)


def critical_spring(ms: float) -> dict:
    """Ressort critique (masse 1, ζ = 1) qui se stabilise à ±1 % en ms millisecondes.

    Entrée : ms — durée visée, strictement positive.
    Sortie : ressort arrondi. Erreurs : ZeroDivisionError si ms est nul.
    Dépendances : CRITICAL_WT, spring.
    """
    w = CRITICAL_WT / (ms / 1000)
    return spring(1, w * w, 2 * w)
```

```python
"""Hyprland 0.56.2 — éléments d'animation et styles acceptés.

Relevé sur la version installée (`hyprctl animations -j`) et dans la source de
CHyprAnimationManager::styleValidInConfigVar. Ces tables sont à revérifier à chaque
mise à jour de Hyprland : tests/test_motion_hyprland.py compare NODES à la version
en cours d'exécution.
"""

from __future__ import annotations

# Tous les éléments publics de l'arbre d'animation, hors __internal_fadeCTM.
NODES = frozenset({
    "global", "windows", "windowsIn", "windowsOut", "windowsMove",
    "layers", "layersIn", "layersOut",
    "fade", "fadeIn", "fadeOut", "fadeSwitch", "fadeShadow", "fadeGlow", "fadeDim",
    "fadeLayers", "fadeLayersIn", "fadeLayersOut", "fadePopups", "fadePopupsIn", "fadePopupsOut", "fadeDpms",
    "border", "borderangle", "shadowangle", "glowangle",
    "workspaces", "workspacesIn", "workspacesOut", "specialWorkspace", "specialWorkspaceIn", "specialWorkspaceOut",
    "zoomFactor", "monitorAdded",
})

# Les parents intermédiaires ne sont pas déclarés, puisque chacun de leurs enfants l'est.
PARENTS = frozenset({"windows", "layers", "fade", "fadeLayers", "fadePopups", "workspaces", "specialWorkspace"})

# Les 26 éléments terminaux, plus global, qui sert de valeur par défaut aux éléments futurs.
REQUIRED = NODES - PARENTS

# Styles qui ne déplacent pas l'élément et ne changent pas son échelle.
STILL = ("", "fade", "loop", "once")


def _percent(style: str) -> bool:
    """Pourcentage final d'un style, par exemple « popin 90% ».

    Entrée : style. Sortie : vrai si le dernier mot, après une espace, est un entier suivi de « % ».
    Erreurs : aucune. Dépendances : aucune.
    """
    head, _, tail = style.rpartition(" ")
    return bool(head) and tail.endswith("%") and tail[:-1].isdigit()


def style_error(leaf: str, style: str) -> str:
    """Validité d'un style pour un élément, selon la règle de Hyprland 0.56.2.

    Entrées : leaf — nom de l'élément ; style — style demandé, vide s'il n'y en a pas.
    Sortie : chaîne vide si le style est accepté, sinon la raison du refus.
    Erreurs : aucune. Dépendances : _percent.
    """
    if not style:
        return ""
    if leaf.startswith("window"):
        if style.startswith("slide") or style in ("gnome", "gnomed"):
            return ""
        if style.startswith("popin"):
            return "" if "%" not in style or _percent(style) else "pourcentage invalide"
        return "style inconnu"
    if leaf.startswith(("workspaces", "specialWorkspace")):
        if style in ("slide", "slidevert", "fade"):
            return ""
        if style.startswith("slide"):
            return "" if "%" not in style or _percent(style) else "pourcentage invalide"
        return "style inconnu"
    if leaf.endswith("angle"):
        return "" if style in ("loop", "once") else "style inconnu"
    if leaf.startswith("layers"):
        if style == "fade" or style.startswith("slide"):
            return ""
        if style.startswith("popin"):
            return "" if "%" not in style or _percent(style) else "pourcentage invalide"
        return "style inconnu"
    return "cet élément n'accepte aucun style"
```

```toml
# motion.toml — source unique du mouvement de DuneI.
#
# Chaque transition est un ressort : masse, raideur, amortissement, sur une progression
# normalisée de 0 à 1 (modèle de Hyprland et du banc proto/motion). Les valeurs ont été
# validées au ressenti dans le banc le 2026-10-13. generate_motion.py en dérive le
# ressort de sortie, le ressort réduit, puis les fichiers de generated/.
version = "0.1.0"

[meta]
name   = "DuneI"
prefix = "dunei"                  # Préfixe des courbes Hyprland, par exemple dunei-state-in.

[rules]                           # Règles chiffrées du chapitre 6 de la bible, vérifiées à chaque génération.
exit_ratio        = 0.70          # Une sortie dure environ 70 % de l'entrée.
exit_tolerance    = 0.10          # Les rapports compris entre 0,60 et 0,80 sont acceptés.
overshoot_allowed = ["window"]    # Seules ces transitions peuvent dépasser leur cible (ζ < 1).
stagger_allowed   = ["reveal"]    # Seules ces transitions peuvent échelonner leurs éléments.
stagger_cap       = 5             # Nombre maximal d'éléments échelonnés dans une cascade.
reduced_fade_ms   = 100           # Durée du fondu qui remplace tout mouvement en mode réduit.

[transitions.state]
usage      = "Changement d'état sur place : survol, focus, pression, sélection."
mass       = 0.50
stiffness  = 1100
dampening  = 50                   # Nom de clé de Hyprland 0.56.2 (D13).
exit_ratio = 0.70

[transitions.reveal]
usage      = "Apparition d'éléments dans le flux."
mass       = 1.00
stiffness  = 600
dampening  = 50
exit_ratio = 0.70
stagger_ms = 40                   # Décalage entre deux éléments successifs de la cascade.

[transitions.overlay]
usage      = "Entrée d'une surface de couche 3."
mass       = 1.00
stiffness  = 400
dampening  = 44
exit_ratio = 0.70

[transitions.window]
usage      = "Ouverture et fermeture de fenêtre."
mass       = 0.70
stiffness  = 300
dampening  = 27
exit_ratio = 0.70

[transitions.reflow]
usage      = "Réagencement du tuilage."
mass       = 0.80
stiffness  = 500
dampening  = 40
symmetric  = true                 # Rien n'entre ni ne sort : le même ressort sert dans les deux sens.

[transitions.spectacle]
usage      = "Moments sans contenu."
mass       = 1.00
stiffness  = 150
dampening  = 25

[transitions.navigate]

windowsMove         = { use = "reflow" }

fadeOut             = { use = "window",    phase = "out" }

layersOut           = { use = "overlay",   phase = "out", style = "popin 96%" }
fadeLayersIn        = { use = "overlay",   phase = "in" }
fadeLayersOut       = { use = "overlay",   phase = "out" }
fadePopupsIn        = { use = "reveal",    phase = "in" }
```

```python
"""Résolution du mouvement — motion.toml → ressorts d'entrée, de sortie et réduit, puis animations Hyprland."""

from __future__ import annotations

import tomllib
from pathlib import Path

from .spring import critical_spring, exit_spring, simulate, spring, zeta


def load(src: Path) -> dict:
    """Lecture de la source de vérité du mouvement.

    Entrée : src — chemin de motion.toml.
    Sortie : dictionnaire brut du TOML, dans l'ordre du fichier.
    Erreurs : FileNotFoundError si absent, tomllib.TOMLDecodeError si invalide.
    Dépendances : tomllib.
    """
    with src.open("rb") as fh:
        return tomllib.load(fh)


def describe(s: dict) -> dict:
    """Ressort accompagné de ses grandeurs indicatives.

    Entrée : s — ressort arrondi.
    Sortie : {mass, stiffness, dampening, zeta, settle_ms}.
    Erreurs : aucune pour un ressort aux valeurs positives. Dépendances : spring.zeta, spring.simulate.
    """
    return {**s, "zeta": round(zeta(s), 2), "settle_ms": simulate(s)[0]}


def _animation(leaf: str, entry: dict, prefix: str) -> dict:
    """Animation du mode normal pour un élément de Hyprland.

    Entrées : leaf — élément ; entry — sa table dans [hyprland] ; prefix — préfixe des courbes.
    Sortie : entrée prête pour hl.animation. Erreurs : KeyError si `use` manque sur un élément actif.
    Dépendances : aucune.
    """
    if entry.get("enabled", True) is False:
        return {"leaf": leaf, "enabled": False}
    a = {"leaf": leaf, "enabled": True, "speed": 1, "spring": f"{prefix}-{entry['use']}-{entry.get('phase', 'in')}"}
    return a | ({"style": entry["style"]} if "style" in entry else {})


def _reduced_animation(leaf: str, entry: dict, override: dict, prefix: str) -> dict:
    """Animation du mouvement réduit pour un élément de Hyprland.

    Entrées : leaf — élément ; entry — sa table dans [hyprland] ;
              override — sa table dans [hyprland_reduced], vide s'il n'y figure pas ; prefix — préfixe.
    Sortie : entrée prête pour hl.animation, sur le ressort réduit et sans autre style que celui de override.
    Erreurs : aucune. Dépendances : aucune.
    """
    if entry.get("enabled", True) is False or override.get("enabled", True) is False:
        return {"leaf": leaf, "enabled": False}
    a = {"leaf": leaf, "enabled": True, "speed": 1, "spring": f"{prefix}-reduced"}
    return a | ({"style": override["style"]} if "style" in override else {})


def resolve(raw: dict) -> dict:
    """Résolution complète d'un motion.toml déjà validé par checks.errors.

    Entrée : raw — sortie de load.
    Sortie : {version, prefix, rules, transitions, reduced, curves, animations, reduced_animations}
             (le détail est donné dans le plan, tâche 3).
    Erreurs : KeyError si raw n'a pas été validé et qu'une clé manque.
    Dépendances : spring, describe, _animation, _reduced_animation.
    """
    prefix, rules = raw["meta"]["prefix"], raw["rules"]
    transitions, curves = {}, {}
    for name, t in raw["transitions"].items():
        s_in = spring(t["mass"], t["stiffness"], t["dampening"])
        s_out = s_in if t.get("symmetric") else exit_spring(s_in, t["exit_ratio"])
        transitions[name] = {"usage": t["usage"], "in": describe(s_in), "out": describe(s_out),
                             "overshoot": t.get("overshoot", False), "symmetric": t.get("symmetric", False)}
        if "stagger_ms" in t:
            transitions[name]["stagger"] = {"ms": t["stagger_ms"], "cap": rules["stagger_cap"]}
        curves[f"{prefix}-{name}-in"], curves[f"{prefix}-{name}-out"] = s_in, s_out
    reduced = critical_spring(rules["reduced_fade_ms"])
    curves[f"{prefix}-reduced"] = reduced
    hyp, overrides = raw["hyprland"], raw["hyprland_reduced"]
    return {
        "version": raw["version"], "prefix": prefix, "rules": rules,
        "transitions": transitions, "reduced": describe(reduced),
        "curves": dict(sorted(curves.items())),
        "animations": [_animation(leaf, e, prefix) for leaf, e in hyp.items()],
        "reduced_animations": [_reduced_animation(leaf, e, overrides.get(leaf, {}), prefix) for leaf, e in hyp.items()],
    }
```

```python
"""Contrôles du mouvement — règles du chapitre 6 de la bible et API de Hyprland 0.56.2.

Une erreur bloque la génération : aucun fichier n'est écrit. Un avertissement est
seulement affiché. errors travaille sur le TOML brut et ne lève jamais d'exception,
afin que chaque défaut soit signalé par un message plutôt que par une trace Python.
"""

from __future__ import annotations

from .hyprland import NODES, REQUIRED, STILL, style_error
from .spring import spring, zeta

VERSION = "0.1.0"
SECTIONS = ("meta", "rules", "transitions", "hyprland", "hyprland_reduced")
META_KEYS = ("name", "prefix")
RULE_KEYS = ("exit_ratio", "exit_tolerance", "overshoot_allowed", "stagger_allowed", "stagger_cap", "reduced_fade_ms")
SPRING_KEYS = ("usage", "mass", "stiffness", "dampening")
REDUCED_FLOOR_MS = 100   # Plancher du fondu réduit fixé par la bible.


def _transitions(raw: dict) -> list[str]:
    """Contrôle des transitions : clés, valeurs positives, dépassement, sortie et cascade.

    Entrée : raw — TOML brut, sections et règles déjà présentes.
    Sortie : liste d'erreurs. Erreurs : aucune exception. Dépendances : spring, zeta.
    """
    rules, err = raw["rules"], []
    lo = round(rules["exit_ratio"] - rules["exit_tolerance"], 6)
    hi = round(rules["exit_ratio"] + rules["exit_tolerance"], 6)
    for name, t in raw["transitions"].items():
        p = f"transitions.{name}"
        if not isinstance(t, dict):
            err.append(f"{p} : doit être une table")
            continue
        needed = SPRING_KEYS + (() if t.get("symmetric") else ("exit_ratio",))
        missing = [k for k in needed if k not in t]
        err += [f"{p} : clé « {k} » manquante" for k in missing]
        nonpositive = [k for k in ("mass", "stiffness", "dampening") if k in t and t[k] <= 0]
        err += [f"{p} : {k} doit être strictement positif" for k in nonpositive]
        if missing or nonpositive:
            continue
        allowed = name in rules["overshoot_allowed"]
        z = zeta(spring(t["mass"], t["stiffness"], t["dampening"]))
        if z < 1 and not allowed:
            err.append(f"{p} : ζ {z:.2f} < 1, dépassement interdit par la bible")
        if t.get("overshoot") and not allowed:
            err.append(f"{p} : overshoot réservé à {', '.join(rules['overshoot_allowed'])}")
        if t.get("symmetric") and "exit_ratio" in t:
            err.append(f"{p} : transition symétrique, exit_ratio interdit")
        elif not t.get("symmetric") and not lo <= t["exit_ratio"] <= hi:
            err.append(f"{p} : exit_ratio {t['exit_ratio']} hors de l'intervalle {lo}–{hi}")
        if "stagger_ms" in t and name not in rules["stagger_allowed"]:
            err.append(f"{p} : cascade interdite, réservée à {', '.join(rules['stagger_allowed'])}")
    return err


def _hyprland(raw: dict) -> list[str]:
    """Contrôle de [hyprland] : éléments connus et complets, transition, phase et style valides.

    Entrée : raw — TOML brut, sections présentes.
    Sortie : liste d'erreurs. Erreurs : aucune exception.
    Dépendances : hyprland.NODES, hyprland.REQUIRED, hyprland.style_error.
    """
    hyp, names = raw["hyprland"], set(raw["transitions"])
    err = [f"hyprland : élément « {leaf} » non déclaré" for leaf in sorted(REQUIRED - set(hyp))]
    for leaf, e in hyp.items():
        p = f"hyprland.{leaf}"
        if leaf not in NODES:
            err.append(f"{p} : élément Hyprland inconnu")
        elif not isinstance(e, dict):
            err.append(f"{p} : doit être une table, par exemple {{ use = \"state\" }}")
        elif e.get("enabled", True) is not False:
            if e.get("use") not in names:
                err.append(f"{p} : transition « {e.get('use')} » inconnue")
            if e.get("phase", "in") not in ("in", "out"):
                err.append(f"{p} : phase « {e['phase']} » invalide, in ou out attendu")
            if why := style_error(leaf, e.get("style", "")):
                err.append(f"{p} : style « {e['style']} » invalide ({why})")
    return err


def _reduced(raw: dict) -> list[str]:
    """Contrôle de [hyprland_reduced] : éléments déclarés, styles valides, aucun mouvement résiduel.

    Entrée : raw — TOML brut, sections présentes.
    Sortie : liste d'erreurs. Erreurs : aucune exception.
    Dépendances : hyprland.STILL, hyprland.style_error.
    """
    hyp, red, err = raw["hyprland"], raw["hyprland_reduced"], []
    for leaf, o in red.items():
        p = f"hyprland_reduced.{leaf}"
        if leaf not in hyp:
            err.append(f"{p} : élément absent de [hyprland]")
        elif not isinstance(o, dict):
            err.append(f"{p} : doit être une table, par exemple {{ style = \"fade\" }}")
        elif why := style_error(leaf, o.get("style", "")):
            err.append(f"{p} : style « {o['style']} » invalide ({why})")
    for leaf, e in hyp.items():
        if isinstance(e, dict) and e.get("enabled", True) is not False and leaf not in red \
                and e.get("style", "") not in STILL:
            err.append(f"hyprland_reduced : {leaf} doit remplacer le style « {e['style']} »")
    return err


def errors(raw: dict) -> list[str]:
    """Contrôles bloquants d'un motion.toml brut.

    Entrée : raw — sortie de model.load.
    Sortie : liste d'erreurs, vide si la source est conforme.
    Erreurs : aucune exception ; chaque défaut devient un message.
    Dépendances : _transitions, _hyprland, _reduced.
    """
    if raw.get("version") != VERSION:
        return [f"version {raw.get('version')} inconnue (attendue : {VERSION})"]
    err = [f"section [{s}] manquante" for s in SECTIONS if not isinstance(raw.get(s), dict)]
    if err:
        return err
    err += [f"meta : clé « {k} » manquante" for k in META_KEYS if k not in raw["meta"]]
    err += [f"rules : clé « {k} » manquante" for k in RULE_KEYS if k not in raw["rules"]]
    if err:
        return err
    if raw["rules"]["reduced_fade_ms"] <= 0:
        err.append("rules : reduced_fade_ms doit être strictement positif")
    return err + _transitions(raw) + _hyprland(raw) + _reduced(raw)


def warnings(model: dict) -> list[str]:
    """Contrôles non bloquants du modèle résolu.

    Entrée : model — sortie de model.resolve.
    Sortie : liste d'avertissements, vide si tout est conforme.
    Erreurs : KeyError si le modèle est incomplet. Dépendances : REDUCED_FLOOR_MS.
    """
    ms = model["rules"]["reduced_fade_ms"]
    return [f"reduced_fade_ms = {ms} ms, sous le plancher de {REDUCED_FLOOR_MS} ms fixé par la bible"] \
        if ms < REDUCED_FLOOR_MS else []
```

```python
"""Cible Hyprland 0.56 — module Lua des ressorts et des animations.

Le module RETOURNE une table, sans appeler hl.curve ni hl.animation lui-même :
config/hypr/modules/motion.lua l'applique, en mode normal ou réduit. Le fichier
se vérifie donc avec un interpréteur Lua nu, sans Hyprland.
"""

from __future__ import annotations

import json


def value(v) -> str:
    """Littéral Lua d'une valeur simple.

    Entrée : v — booléen, nombre ou chaîne.
    Sortie : texte Lua ; les nombres sont écrits sans zéro superflu (1100, 0.5, 2244.9).
    Erreurs : aucune. Dépendances : json, pour l'échappement des chaînes.
    """
    if isinstance(v, bool):
        return "true" if v else "false"
    if isinstance(v, (int, float)):
        return f"{v:.10g}"
    return json.dumps(v, ensure_ascii=False)


def table(d: dict) -> str:
    """Table Lua sur une ligne, dans l'ordre des clés.

    Entrée : d — dictionnaire de valeurs simples.
    Sortie : texte « { clé = valeur, … } ». Erreurs : aucune. Dépendances : value.
    """
    return "{ " + ", ".join(f"{k} = {value(v)}" for k, v in d.items()) + " }"


def emit(model: dict) -> str:
    """Module Lua des courbes et des deux jeux d'animations.

    Entrée : model — sortie de model.resolve.
    Sortie : contenu de motion.lua (une table Lua retournée).
    Erreurs : KeyError si le modèle est incomplet. Dépendances : table.
    """
    return "\n".join([
        "-- GÉNÉRÉ depuis motion.toml par generate_motion.py. Ne pas modifier à la main.",
        '-- Usage : require("modules.motion").apply(false)',
        "return {",
        "    -- Courbes : un ressort d'entrée et un ressort de sortie par transition, plus le fondu réduit.",
        "    curves = {",
        *(f'        ["{name}"] = {table({"type": "spring", **s})},' for name, s in model["curves"].items()),
        "    },",
        "    -- Animations du mode normal, une entrée par élément de Hyprland, global en premier.",
        "    animations = {",
        *(f"        {table(a)}," for a in model["animations"]),
        "    },",
        "    -- Animations du mouvement réduit : mêmes éléments, fondu court, sans déplacement ni changement d'échelle.",
        "    reduced = {",
        *(f"        {table(a)}," for a in model["reduced_animations"]),
        "    },",
        "}",
    ]) + "\n"
```

```python
#!/usr/bin/env python3
"""Générateur du mouvement — point d'entrée.

Lit motion.toml, vérifie les règles du chapitre 6 de la bible et l'API d'animation
de Hyprland, puis produit les fichiers consommés par le compositeur et la coque.
La logique vit dans le paquet dunei_motion.

    python3 generate_motion.py [--src motion.toml] [--out generated]
"""

from __future__ import annotations

import argparse
import sys
import tomllib
from pathlib import Path

from dunei_motion.checks import errors, warnings
from dunei_motion.emitters import TARGETS
from dunei_motion.model import load, resolve


def main() -> int:
    """Génération de toutes les cibles du mouvement.

    Entrées : arguments de ligne de commande --src et --out.
    Sortie : code de retour 0 si succès, 1 si la source est absente, illisible ou non conforme.
    Erreurs : messages sur la sortie d'erreur ; aucun fichier n'est écrit en cas d'erreur.
    Dépendances : dunei_motion (model, checks, emitters).
    """
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--src", default=Path("motion.toml"), type=Path)
    ap.add_argument("--out", default=Path("generated"), type=Path)
    args = ap.parse_args()

    try:
        raw = load(args.src)
    except (OSError, tomllib.TOMLDecodeError) as e:
        print(f"source illisible : {args.src} ({e})", file=sys.stderr)
        return 1

    if errs := errors(raw):
        for e in errs:
            print(f"  erreur : {e}", file=sys.stderr)
        print(f"\n{len(errs)} erreur(s) : aucun fichier écrit.", file=sys.stderr)
        return 1

    model = resolve(raw)
    for w in warnings(model):
        print(f"  avertissement : {w}", file=sys.stderr)

    args.out.mkdir(parents=True, exist_ok=True)
    for filename, emitter in TARGETS.items():
        (args.out / filename).write_text(emitter(model), encoding="utf-8")
        print(f"  écrit {args.out / filename}")

    print(f"\nMouvement généré : {len(model['transitions'])} transitions vers {len(TARGETS)} cibles.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```lua
-- Module de mouvement : applique dans Hyprland les ressorts générés depuis motion.toml.
-- Aucune valeur n'est écrite ici ; elles viennent toutes de generated/motion.lua.
local motion = require("generated.motion")
local M = {}

-- apply : déclare les courbes, puis le jeu d'animations normal ou réduit.
-- Entrée : reduced (booléen), vrai pour le mouvement réduit.
-- Sortie : aucune. Erreurs : celles de hl.curve et hl.animation.
-- Dépendances : generated.motion, API hl de Hyprland 0.56.
-- Un appel ultérieur, depuis le fragment machine par exemple, remplace le précédent.
function M.apply(reduced)
    for name, curve in pairs(motion.curves) do hl.curve(name, curve) end
    for _, animation in ipairs(reduced and motion.reduced or motion.animations) do hl.animation(animation) end
end

return M
```
