> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Produire `fonts.conf` depuis `tokens.toml`, le déployer avec `make install` dans
`$XDG_CONFIG_HOME/fontconfig/conf.d/50-dunei.conf`, et prouver avec le vrai `fc-match` que les noms
canoniques, les familles génériques, la chaîne de repli et le rendu sont résolus comme la bible le veut.

**Architecture:** Une huitième cible de `dunei_theme` (`emitters/fontconfig.py`) lit `[type]`,
`[type.installed]` et `[type.rendering]`. `deploy.py` passe d'une destination à un registre
`DESTINATIONS` (`hypr`, `fontconfig`) dont toutes les entrées sont planifiées avant toute action. Un
test d'intégration interroge `fc-match` avec un `XDG_CONFIG_HOME` temporaire.

**Tech Stack:** Python 3 (bibliothèque standard : `tomllib`, `xml.etree`, `xml.sax.saxutils`,
`unittest`, `subprocess`), fontconfig 2.18 (`fc-match`, `fc-list`), make.

```python
"""Cible fontconfig — XML valide, noms et rendu tirés des jetons, valeurs invalides refusées."""

import copy
import subprocess
import sys
import tempfile
import unittest
import xml.etree.ElementTree as ET
from pathlib import Path

from dunei_theme.emitters.fontconfig import emit
from dunei_theme.model import load, resolve

ROOT = Path(__file__).resolve().parent.parent
MARK = "GÉNÉRÉ depuis tokens.toml — NE PAS MODIFIER À LA MAIN"


def aliases(text: str) -> list[tuple[str, list[str]]]:
    """Alias du fichier, dans l'ordre.

    Entrée : text — contenu de fonts.conf. Sortie : [(famille demandée, [familles préférées])].
    Erreurs : ET.ParseError si le XML est invalide. Dépendances : xml.etree.
    """
    root = ET.fromstring(text.encode("utf-8"))
    return [(a.find("family").text, [f.text for f in a.find("prefer")]) for a in root.iter("alias")]


class FontconfigEmitTest(unittest.TestCase):
    """Contenu de fonts.conf, recalculé depuis le modèle."""

    def setUp(self):
        self.model = resolve(load(ROOT / "tokens.toml"))
        self.t = self.model["raw"]["type"]

    def test_header(self):
        lines = emit(self.model).splitlines()
        self.assertTrue(lines[0].startswith("<?xml"))
        self.assertIn(MARK, lines[1])
        self.assertIn("<!DOCTYPE fontconfig", lines[2])

    def test_aliases_in_order(self):
        t, chain = self.t, self.t["fallback_chain"]
        self.assertEqual(aliases(emit(self.model)), [
            ("sans-serif", [t["interface"]]),
            ("monospace", [t["mono"]]),
            (t["interface"], [t["installed"]["interface"], *chain]),
            (t["mono"], [t["installed"]["mono"], *chain]),
        ])

    def test_serif_untouched(self):
        self.assertNotIn("serif", [f for f, _ in aliases(emit(self.model))])

    def test_rendering(self):
        root = ET.fromstring(emit(self.model).encode("utf-8"))
        edits = {e.get("name"): e[0].text for e in root.find("match").iter("edit")}
        r = self.t["rendering"]
        self.assertEqual(edits, {"antialias": str(r["antialias"]).lower(),
                                 "hintstyle": r["hintstyle"], "rgba": r["rgba"]})

    def test_escapes_family_names(self):
        model = copy.deepcopy(self.model)
        model["raw"]["type"]["installed"]["interface"] = "Fonte & Cie"
        self.assertIn((self.t["interface"], ["Fonte & Cie", *self.t["fallback_chain"]]), aliases(emit(model)))

    def test_refuses_bad_values(self):
        for key, value in (("hintstyle", "hintslite"), ("rgba", "rbg"), ("antialias", "oui")):
            with self.subTest(key=key):
                model = copy.deepcopy(self.model)
                model["raw"]["type"]["rendering"][key] = value

            self.assertIn("hintstyle", res.stderr)
            self.assertFalse(out.exists())
```

```python
"""Cible fontconfig — indirection des noms de polices, chaîne de repli et rendu.

Les jetons nomment les polices par leur nom canonique ; les paquets les installent sous
d'autres noms, et une famille introuvable retombe silencieusement sur la police système.
Ce fichier fait la correspondance une seule fois, pour toutes les applications qui passent
par fontconfig.

fontconfig applique les règles dans l'ordre, chacune voyant le résultat des précédentes.
Les familles génériques sont donc redirigées vers les noms canoniques avant que ceux-ci
le soient vers les familles installées : une seule passe suffit.

Déployé dans $XDG_CONFIG_HOME/fontconfig/conf.d/50-dunei.conf, lu avant les préférences
d'Arch (60-*.conf).
"""

from __future__ import annotations

from xml.sax.saxutils import escape

HINTSTYLES = ("hintnone", "hintslight", "hintmedium", "hintfull")
RGBA = ("none", "rgb", "bgr", "vrgb", "vbgr", "unknown")
GENERICS = {"sans-serif": "interface", "monospace": "mono"}  # Famille générique → rôle de [type].


def check_rendering(rendering: dict) -> None:
    """Contrôle des valeurs de rendu, avant toute écriture.

    Entrée : rendering — table [type.rendering] du TOML.
    Sortie : aucune.
    Erreurs : ValueError, qui nomme la clé fautive, si antialias n'est pas un booléen ou si
              hintstyle ou rgba n'est pas une constante de fontconfig ; KeyError si une clé manque.
    Dépendances : HINTSTYLES, RGBA.
    """
    if not isinstance(rendering["antialias"], bool):
        raise ValueError(f"[type.rendering] antialias : booléen attendu, reçu {rendering['antialias']!r}")
    for key, allowed in (("hintstyle", HINTSTYLES), ("rgba", RGBA)):
        if rendering[key] not in allowed:
            raise ValueError(f"[type.rendering] {key} : {rendering[key]!r} n'est pas l'une de {', '.join(allowed)}")


def alias(family: str, prefer: list[str]) -> str:
    """Bloc <alias> : les familles de prefer sont insérées, dans l'ordre, devant family.

    Entrées : family — famille demandée ; prefer — familles à lui préférer.
    Sortie : texte XML indenté. La liaison « same » reprend la force de la demande d'origine.
    Erreurs : aucune. Dépendances : xml.sax.saxutils.escape.
    """
    fams = "".join(f"<family>{escape(f)}</family>" for f in prefer)
    return f'  <alias binding="same">\n    <family>{escape(family)}</family>\n    <prefer>{fams}</prefer>\n  </alias>\n'


def emit(model: dict) -> str:
    """Configuration fontconfig de DuneI.

    Entrée : model — sortie de model.resolve.
    Sortie : contenu de fonts.conf : rendu, familles génériques, noms canoniques.
    Erreurs : ValueError si une valeur de rendu est refusée (check_rendering) ; KeyError si une
              clé de [type], [type.installed] ou [type.rendering] manque.
    Dépendances : check_rendering, alias, GENERICS.
    """
    t = model["raw"]["type"]
    r = t["rendering"]
    check_rendering(r)
    out = [
        '<?xml version="1.0"?>\n',
        "<!-- GÉNÉRÉ depuis tokens.toml — NE PAS MODIFIER À LA MAIN -->\n",
        '<!DOCTYPE fontconfig SYSTEM "urn:fontconfig:fonts.dtd">\n',
        "<fontconfig>\n",
        "  <!-- Rendu : écran dense, niveaux de gris, hinting léger. -->\n",
        '  <match target="font">\n',
        f'    <edit name="antialias" mode="assign"><bool>{str(r["antialias"]).lower()}</bool></edit>\n',
        f'    <edit name="hintstyle" mode="assign"><const>{r["hintstyle"]}</const></edit>\n',
        f'    <edit name="rgba" mode="assign"><const>{r["rgba"]}</const></edit>\n',
        "  </match>\n",
        "  <!-- Familles génériques vers les noms canoniques. -->\n",
        *(alias(g, [t[role]]) for g, role in GENERICS.items()),
        "  <!-- Noms canoniques vers les familles installées, puis la chaîne de repli. -->\n",
        *(alias(t[role], [t["installed"][role], *t["fallback_chain"]]) for role in GENERICS.values()),
        "</fontconfig>\n",
    ]
    return "".join(out)
```

```python
def plan_fontconfig(repo: Path, home: Path) -> tuple[list[tuple], list[str]]:
    """Inspection de la destination fontconfig, sans aucune modification.

    Entrées : repo — racine du dépôt ; home — dossier de configuration racine.
    Sortie : (actions dans l'ordre d'exécution, erreurs).
    Erreurs : aucune exception ; source absente, dossier qui n'en est pas un, fichier ou lien
              étranger à la place de 50-dunei.conf deviennent des messages.
    Dépendances : _ours, FONTCONFIG.
    """
    src, dest = repo / "generated" / FONTCONFIG[0], home / "fontconfig" / FONTCONFIG[1]
    actions, err = [], []
    if not src.exists():
        err.append(f"source manquante : {src} (lancer make tokens)")
    for d in (dest.parent.parent, dest.parent):
        if d.is_symlink() or (d.exists() and not d.is_dir()):
            return actions, err + [f"{d} existe mais n'est pas un dossier réel"]
        if not d.exists():
            actions.append(("mkdir", d))
    if dest.is_symlink() or (dest.exists() and not _ours(dest)):
        err.append(f"{dest} existe mais n'a pas été posé par DuneI")
    return actions + [("copy", src, dest)], err


DESTINATIONS = {"hypr": plan_hypr, "fontconfig": plan_fontconfig}  # Nom → planification.


def plan(repo: Path, home: Path) -> tuple[list[tuple], list[str]]:
    """Inspection de toutes les destinations, sans aucune modification.

    Entrées : repo — racine du dépôt ; home — dossier de configuration racine.
    Sortie : (actions de toutes les destinations, dans l'ordre de DESTINATIONS ; erreurs).
             Une seule erreur suffit à tout refuser.
    Erreurs : aucune exception ; chaque problème devient un message. Dépendances : DESTINATIONS.
    """
    actions, err = [], []
    if (repo / ".git").is_file():
        err.append(f"{repo} est un arbre de travail secondaire : les liens disparaîtraient avec lui, "
                   "lancer make install depuis le dépôt principal")
    for plan_one in DESTINATIONS.values():
        a, e = plan_one(repo, home)
        actions, err = actions + a, err + e
    return actions, err
```

Note : `apply` crée les dossiers avec `mkdir(parents=True)` ; quand `fontconfig/` manque, les deux actions `mkdir` se suivent et la seconde trouve son parent. Aucune modification de `apply` n'est nécessaire.

Dans `main`, `dest = config_dir(os.environ)` devient `home = config_home(os.environ)`, `plan(REPO, dest)` devient `plan(REPO, home)`, et `dest` devient `home` dans les deux messages de synthèse. Dans la docstring de `main`, `Dépendances : config_dir, plan, …` devient `Dépendances : config_home, plan, …`.


```python
"""Intégration — le fonts.conf produit, interrogé par le vrai fc-match dans un dossier de configuration temporaire.

La configuration de la session n'est jamais touchée : XDG_CONFIG_HOME pointe vers un dossier jetable
qui ne contient que fontconfig/conf.d/50-dunei.conf, lu par /etc/fonts/conf.d/50-user.conf.
"""

import os
import shutil
import subprocess
import tempfile
import unittest
from pathlib import Path

from dunei_theme.emitters.fontconfig import HINTSTYLES, emit
from dunei_theme.model import load, resolve

ROOT = Path(__file__).resolve().parent.parent
FC_MATCH, FC_LIST = shutil.which("fc-match"), shutil.which("fc-list")
RGBA = {"unknown": 0, "rgb": 1, "bgr": 2, "vrgb": 3, "vbgr": 4, "none": 5}  # Constantes FC_RGBA_*.
HIRAGANA, EMOJI, SYMBOL = 0x3042, 0x1F600, 0x1FB00


@unittest.skipUnless(FC_MATCH and FC_LIST, "fontconfig absent")
class FontconfigMatchTest(unittest.TestCase):
    """Les critères 1 à 3 de la spec, mesurés sur les polices réellement installées."""

    @classmethod
    def setUpClass(cls):
        cls.tmp = tempfile.TemporaryDirectory()
        conf = Path(cls.tmp.name) / "fontconfig" / "conf.d" / "50-dunei.conf"
        conf.parent.mkdir(parents=True)
        model = resolve(load(ROOT / "tokens.toml"))
        conf.write_text(emit(model), encoding="utf-8")
        cls.t = model["raw"]["type"]
        cls.env = {**os.environ, "XDG_CONFIG_HOME": cls.tmp.name}

    @classmethod
    def tearDownClass(cls):
        cls.tmp.cleanup()

    def fc(self, tool: str, *args: str) -> str:
        """Sortie d'un outil fontconfig sous la configuration de test.

        Entrées : tool — chemin de fc-match ou fc-list ; args — arguments.
        Sortie : sortie standard. Erreurs : CalledProcessError si l'outil échoue.
        Dépendances : subprocess.
        """
        return subprocess.run([tool, *args], capture_output=True, text=True, env=self.env, check=True).stdout

    def first(self, query: str) -> str:
        """Première famille choisie par fc-match. Entrée : motif fontconfig. Sortie : nom de famille."""
        return self.fc(FC_MATCH, "-f", "%{family[0]}", query)

    def test_fonts_installed(self):
        for fam in [*self.t["installed"].values(), *self.t["fallback_chain"]]:
            with self.subTest(family=fam):
                self.assertTrue(self.fc(FC_LIST, fam, "family").strip(),
                                f"{fam} n'est pas installée : voir les paquets du carnet, section Typographie")

    def test_canonical_and_generic_names(self):
        inst = self.t["installed"]
        for query, fam in ((self.t["interface"], inst["interface"]), ("sans-serif", inst["interface"]),
                           (self.t["mono"], inst["mono"]), ("monospace", inst["mono"])):
            with self.subTest(query=query):
                self.assertEqual(self.first(query), fam)

    def test_fallback_chain(self):
        for base in (self.t["interface"], self.t["mono"]):
            for cp in (HIRAGANA, EMOJI, SYMBOL):
                with self.subTest(base=base, cp=hex(cp)):
                    expected = next(f for f in self.t["fallback_chain"] if self.fc(FC_LIST, f"{f}:charset={cp:x}", "family").strip())
                    self.assertEqual(self.first(f"{base}:charset={cp:x}"), expected)

    def test_rendering(self):
        r = self.t["rendering"]
        self.assertEqual(self.fc(FC_MATCH, "-f", "%{antialias} %{hintstyle} %{rgba}", self.t["interface"]),
                         f"{r['antialias']} {HINTSTYLES.index(r['hintstyle'])} {RGBA[r['rgba']]}")

    def test_serif_untouched(self):
        self.assertNotIn(self.first("serif"), self.t["installed"].values())


if __name__ == "__main__":
    unittest.main()
```

Note : `HINTSTYLES` suit l'ordre des constantes `FC_HINT_NONE` (0) à `FC_HINT_FULL` (3) ; `str(True)` donne `True`, comme `%{antialias}`.
